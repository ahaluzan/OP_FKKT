def preberi_covid(ime, sl):
    f = open(ime, encoding="utf8")
    f.readline()

    sez = []

    for vrstica in f:
        drz, reg, primeri, smrti =  vrstica.strip().split(",")
        if drz in sl:
            pop = sl[drz]
        else:
            pop = None

        sez.append((drz,
                    reg,
                    int(primeri),
                    int(smrti),
                    pop))
    return sez   
    


def primerov_na_regijo(sez):
    primerov_regije = {}
    pop_regije = {}

    for _, reg, primeri, _, pop in sez:
        if pop != None:
            if reg not in primerov_regije:
                primerov_regije[reg] = primeri    
                pop_regije[reg] = pop
            else:
                primerov_regije[reg] += primeri
                pop_regije[reg] += pop
    sl = {}

    for reg in primerov_regije:
        sl[reg] = round(primerov_regije[reg]/pop_regije[reg],4)

    return sl

def najslabsa_regija(slovar):
    naj_st = 0
    naj_reg = None

    for reg, st in slovar.items():
        if st > naj_st:
            naj_st = st
            naj_reg = reg

    return naj_reg
