prejemniki = eval(input("Vnesite seznam prejemnikov torte: "))
redni_studenti = eval(input("Vnesite seznam rednih študentov: "))

placniki = []
ze_prejeli = []
for vpisna in prejemniki:
    if vpisna not in ze_prejeli:
        ze_prejeli += [vpisna]
        if vpisna not in redni_studenti:
            placniki += [vpisna]

print(placniki)

kosov = int(input("Koliko kosov torte je na voljo? "))

while kosov > 0:
    st = int(input("Koliko kosov torte želiš? "))
    if st <= kosov:
        kosov -= st
        print("Preostalih kosov: ", kosov)
    else:
        print("Dobiš le", kosov, "kos(e/a/ov).")
        kosov = 0
print("Torte je zmanjkalo!")

def lucasova_stevila(n):
    L_n_2 = 2
    L_n_1 = 1
    stevila = [2, 1]
    while L_n_1 + L_n_2 < n:
        stevila += [L_n_1 + L_n_2]
        L_n_2, L_n_1 = L_n_1, L_n_1 + L_n_2
    return stevila
