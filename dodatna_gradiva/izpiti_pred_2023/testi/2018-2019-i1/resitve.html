def preberi_filme(datoteka):
    with open(datoteka, "r", encoding="utf-8") as f:
        vrstice = f.read().strip().splitlines()
        filmi = []
        for vrstica in vrstice:
            p = vrstica.strip().split(",")
            p[2] = int(p[2])
            ure, minute, sekunde = p[3].split(":")
            ure = float(ure)
            minute = float(minute)
            sekunde = float(sekunde)
            
            p[3] = round(60 * ure + minute + sekunde / 60, 2)
            filmi.append(p)

        return filmi

def povprecna_dolzina(filmi, zanr):
    stevec = 0
    skupaj = 0
    for naslov, zanr_filma, ocena, trajanje in filmi:
        if zanr_filma == zanr:
            skupaj += trajanje
            stevec += 1
    
    if skupaj == 0:
        return 0
    return round(skupaj / stevec, 2)

def presek(ocene, uporabnik1, uporabnik2):
    skupne = []    
    for i in range(len(ocene[uporabnik1])):
        if ocene[uporabnik1][i] > 0 and ocene[uporabnik2][i] > 0:
            skupne.append(i)
    
    return skupne

def ocene_po_filmih(ocene, filmi):
    rezultat = {}
    for film, zanr, ocena, trajanje in filmi:
        rezultat[film] = {}
    
    for uporabnik, ocene_filmov in ocene.items():
        for i in range(len(filmi)):
            ocena = ocene_filmov[i]
            ime_filma = filmi[i][0]
            if ocena > 0:
                rezultat[ime_filma][uporabnik] = ocena

    return rezultat

def najboljsi_film(glasovi):
    najboljsi = None
    najboljsa_ocena = 0
    
    for film, ocene in glasovi.items():
        povprecje = sum(ocene.values()) / len(ocene)
        
        if povprecje > najboljsa_ocena:
            najboljsa_ocena = povprecje
            najboljsi = film
            
    return najboljsi
