def preberi_podatke(datoteka):
   with open(datoteka, "r", encoding="utf-8") as h:
        lines = h.read().strip().split('\n')

        data = {}
        for row in lines[1:-1]:           
            split = row.strip().split(",")
            data[split[0]] = split[1:]

        return data, lines[-1].strip().split(",")[1:]

def oceni_naloge(poskusi, resitve):
    tocke = {}
    for vpisna, odgovori in poskusi.items():
        popravek = []       
        for poskus, resitev in zip(odgovori, resitve):
            if poskus == resitev:
                popravek.append(1)
            elif poskus == "-":
                popravek.append(0)
            else:
                popravek.append(-0.25)
                
        tocke[vpisna] = popravek
    
    return tocke

def ocena(tocke):
    if tocke >= 9:
        ocena = 10
    elif tocke >= 8:
        ocena = 9
    elif tocke >= 7:
        ocena = 8
    elif tocke >= 6:
        ocena = 7
    elif tocke >= 5:
        ocena = 6
    else:
        ocena = 5
        
    return ocena 

def izracunaj_ocene(tocke):
    ocene = {}
    for vpisna, popravki in tocke.items():
        ocene[vpisna] = ocena(sum(popravki))
    return ocene

def porazdelitev_ocen(ocene):
