def preberi_podatke(datoteka):
    with open(datoteka, 'r') as f:
        podatki = []
        for vrstica in f.readlines()[1:]:
            datum, temp, dez, sonce, nevihta = vrstica.strip().split(',')
            leto, mesec, dan = datum.split('-')
            podatki.append(((int(leto), int(mesec), int(dan)),
                        float(temp), float(dez), float(sonce), nevihta.lower() == "da"))
    return podatki

def kolicina_padavin(podatki):
    padavine = {}
    for (leto, mesec, dan), temp, dez, sonce, nevihta in podatki:
        if mesec in padavine:
            padavine[mesec] += dez
        else:
            padavine[mesec] = dez

    return padavine

def za_izlet_primeren(temp, sonce, dez, nevihta):
    return temp > 25 and sonce > 10.0 and dez == 0 and not nevihta
    
    
def stevilo_dni_za_izlete(podatki):
    count = 0
    for datum, temp, dez, sonce, nevihta in podatki:
        if za_izlet_primeren(temp, sonce, dez, nevihta):
            count += 1
    return count

def najvec_padavin(podatki):
    stevilo_dni = 10 
    naj = 0.0

    for start in range(len(podatki) - stevilo_dni):
        stop = start + stevilo_dni

        skupaj = 0.0
        for i in range(start, stop):
            datum, temp, dez, sonce, nevihta = podatki[i]
            skupaj += dez

        if skupaj > naj:
            naj = skupaj

    return naj

def obdobja_za_izlete(podatki):
    intervali = []
    start = None
    stop = None
    
    for datum, temp, dez, sonce, nevihta in podatki:
        if za_izlet_primeren(temp, sonce, dez, nevihta):
            if start is None:
                start = datum
            stop = datum
        else:
            if start:
                intervali.append((start, stop))
                start = None
                stop = None
    
    if start:
        intervali.append((start, podatki[-1][0]))

    return intervali
