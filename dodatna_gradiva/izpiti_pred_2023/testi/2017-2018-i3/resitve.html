def preberi_podatke(datoteka):
    f = open(datoteka, encoding='utf-8')
    vrstice = f.readlines()
    f.close()
    s = []
    for vrstica in vrstice[1:]:
        p = vrstica.strip().split('\t')
        for i in range(1, 7):
            if p[i] != '':
                p[i] = int(p[i])
            else:
                p[i] = None
        s.append([p[0], p[1], p[2], (p[3], p[4]), (p[5], p[6]), p[7]])
    return s

def razlika_v_mesecih(d1, d2):
    if None in d1 or None in d2:
        return None
    else:
        return (d2[0] - d1[0]) * 12 + d2[1] - d1[1]

def najvec_izbruhov(podatki, vei, leto1, leto2):
    s = {}
    for p in podatki:
        vei2=p[2]
        if vei2 == None:
            vei2 = 0
        if p[3][0] >= leto1 and p[3][0] <= leto2 and vei2 >= vei:
            if p[0] not in s:
                s[p[0]] = 0
            s[p[0]]+=1
    if len(s) == 0:
        return (0, set())
    naj = max(s.values())
    mn = set()
    for v in s:
        if s[v] == naj:
            mn.add(v)
    return (naj,mn)

def brez_izbruha(mn, podatki, spodnja_meja, zgornja_meja):
    for p in podatki:
        if p[4][0] == None:
            continue
        if p[3][0] <= zgornja_meja and  p[4][0] >= spodnja_meja :
            if p[0] in mn:
                mn.remove(p[0])
    return mn
