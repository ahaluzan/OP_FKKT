def preberi_podatke(datoteka):
    with open(datoteka, 'r', encoding='utf-8') as h:
        vrstice = h.readlines()
    
    podatki = []
    for vrstica in vrstice[1:]:
        datum, bruto, neto = vrstica.strip().split(',')
        bruto, neto = float(bruto), float(neto)
        leto, mesec = datum.split('M')
        leto, mesec = int(leto), int(mesec)
        podatki.append(((leto, mesec), bruto, neto))
        
    return podatki

def stopnje_obdavcitev(podatki):
    rezultat = []
    for ((leto, mesec), bruto, neto) in podatki:
        obdavcitev = (bruto - neto) / bruto
        rezultat.append(((leto, mesec), obdavcitev))
    
    return rezultat

def drseca_bruto_placa(podatki):
    rezultat = []
    for i in range(len(podatki) - 2):
        val = (podatki[i][1] + podatki[i+1][1] + podatki[i+2][1]) / 3.0
        rezultat.append((podatki[i][0], val))
    
    return rezultat

def povprecne_mesecne_place(podatki):
    vsote = {}
    stevci = {}
    for ((_, mesec), bruto, neto) in podatki:
        if mesec in vsote:
            vsote[mesec] += bruto
            stevci[mesec] += 1
        else:
            vsote[mesec] = bruto
            stevci[mesec] = 1
        
    rezultat = {}

    for mesec, vsota in vsote.items():
        rezultat[mesec] = vsota / stevci[mesec]

    return rezultat

def najboljse_polletje(podatki):
    best_start, best_end, best = None, None, 0
    
    for start in range(len(podatki) - 5):
        stop = start + 5
        
        total = 0
        for i in range(start, stop + 1):
            total += podatki[i][1]
        
        average = total / 6
            
        if average > best:
            best = average
            best_start = start
            best_stop = stop
        
    return podatki[best_start][0], podatki[best_stop][0], best
