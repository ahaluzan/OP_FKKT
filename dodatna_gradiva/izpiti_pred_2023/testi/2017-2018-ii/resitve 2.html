def preberi_podatke(datoteka):
    f = open(datoteka, encoding='utf-8')
    vrstice=f.readlines()
    f.close()
    poleti = []
    for vrstica in vrstice:
        p = vrstica.split(';')
        ocene = []
        for podatek in p[4:9]:
            ocene.append(float(podatek))
        poleti.append( [int(p[0]), p[1], p[2], float(p[3]), tuple(ocene), int(p[9]), float(p[10])] )
    return poleti

def izracun_tock(polet):
    t1 = (polet[3] - 100) * 1.2
    t2 = sum(polet[4]) - min(polet[4]) - max(polet[4])
    t3 = (polet[5] - 8) * -9.3312
    if polet[6] > 0:
        t4 = -polet[6] * 14.40
    else:
        t4 = -polet[6] * 17.42
    return max(round(t1 + t2 + t3 + t4, 1), 0)

def mesto(skakalci, skakalec):
    tocke = skakalci[skakalec]
    uvrstitev = 1
    for podatki in skakalci.items():
        if podatki[1] > tocke:
            uvrstitev+=1
    return uvrstitev

def dolgi_poleti(poleti, najmanj):
    mn = set()
    for polet in poleti:
        if polet[3] >= najmanj:
           mn.add(polet[1])
    return mn

def povprecna_dolzina_po_drzavah(poleti):
    dolzine = {}
    for polet in poleti:
        drzava = polet[2]
        dolzina =  polet[3]
        if drzava not in dolzine:
            dolzine[drzava] = []
        dolzine[drzava] += [dolzina]     
    povpr = {}
    for drzava, dolz in dolzine.items():
        povpr[drzava] = round( sum(dolz) / len(dolz), 1)

    return povpr
