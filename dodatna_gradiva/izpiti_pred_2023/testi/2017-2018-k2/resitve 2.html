def preberi_podatke(dat):
    s = []
    with open(dat, encoding ='utf-8') as f:
        for line in f.readlines():
            if not line[0] == '#':
                s.append(line.strip())     
    return s

def zdruzi_obmocja(s):
    obmocja = {}
    for tablica in s:
        ob, st = tablica.split(' ')
        if ob not in obmocja:
            obmocja[ob] = set()
        obmocja[ob].add(st)
    return obmocja

def pogosta_vozila(s, n):
    pogosto = {}
    for tablica in s:
        if tablica not in pogosto:
            pogosto[tablica] = 0
        pogosto[tablica] += 1

    mn = set()
    for tablica, st in pogosto.items():
        if st >= n:
            mn.add(tablica)

    return mn

def dolzina_poti(letalisca, razdalje, itinerar):
    dolzina = 0
    for i in range(len(itinerar) - 1):
        dolzina += razdalje[letalisca.index(itinerar[i])][letalisca.index(itinerar[i+1])]
    return dolzina
