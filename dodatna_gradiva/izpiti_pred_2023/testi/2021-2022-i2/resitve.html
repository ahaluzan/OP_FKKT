def preberi_vreme(ime):
    sl = {}

    f = open(ime, encoding="utf8")
    f.readline()

    for vrstica in f:
        dat, temp, hum, wind, cond = vrstica.strip().split(";")
        mes, dan, let = dat.split("/")
        sl[(int(dan), int(mes), int(let))] = (float(temp),
                                              float(hum),
                                              float(wind),
                                              cond)
    return sl

def povprecje_po_dnevih(sl, sel_mes = None):
    sez = []
    for dan, mes, let in sl:
        if mes == sel_mes or sel_mes == None:
            sez.append(sl[(dan, mes, let)][0])

    if sez:
        return round(sum(sez)/len(sez),2)

    return False
                       
    
def lepi_dnevi(sl):
    mn = set()
    for dan, (temp, _, _, cond) in sl.items():
        if (temp >= 10 and "Rain" not in cond) or (temp >= 5 and cond == "Clear"):
            mn.add(dan)
    return mn


            
def najvec_soncnih(sl):
    sl_dni = {}
    for (_, mes, _), (_, _, _, cond) in sl.items():
        if cond == "Clear":
            if mes not in sl_dni:
                sl_dni[mes] = 1
            else:
                sl_dni[mes] += 1

    naj_mes = False
    naj_st = 0

    for mes, st in sl_dni.items():
        if st > naj_st:
            naj_st = st
            naj_mes = mes

    return naj_mes
