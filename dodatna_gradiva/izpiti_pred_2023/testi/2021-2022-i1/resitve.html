def preberi_stevec(ime, min_up=0):
    sez = []
    f = open(ime, encoding="utf8")
    f.readline()

    for vr in f:
        date, pct_up, pes, kolo, avto, tov = vr.strip().split(",")
        if float(pct_up) >= min_up:
            sez.append((date,
                        float(pct_up),
                        float(pes),
                        float(kolo),
                        float(avto),
                        float(tov)))
    f.close()
    return sez
                   


def razbij_datum(datum):
    d,t = datum.split("T")
    h = int(t.split(":")[0])

    return d,h
                 

 
def obdelaj_datume(sez):
    sez2 = []
    """
    for date, pct_up, pes, kolo, avto, tov in sez:
        d,h = razbij_datum(date)
        sez2.append((d, h, pct_up, pes, kolo, avto, tov))
    """
    for terka in sez:
        d,h = razbij_datum(terka[0])
        sez2.append((d,h) + terka[1:])
    return sez2
        


def vozil_na_dan(sez):
    sl = {}
    for terka in sez:
        d = terka[0]
        vozil = terka[-1] + terka[-2]
        if d in sl:
            sl[d] += vozil
        else:
            sl[d] = vozil
    return sl

def obremenjeni_dnevi(sl, meja=None):
    mn = set()
    if meja == None:
        #izracunaj povprecje
        vozila = sl.values()
        if len(vozila) > 0:
            meja = sum(vozila)/len(vozila)

    for dan, st in sl.items():
        if st > meja:
            mn.add(dan)

    return mn
