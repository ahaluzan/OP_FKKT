def povprecje(sez):
    vsota = 0
    stevilo = 0
    for el in sez:
        if el >= 0:
            vsota += el
            stevilo += 1
    if stevilo:
        return vsota / stevilo
    return False

def povprecje_interval(sez, zacetek, konec):
    sez = sez[zacetek - 1:konec]
    return povprecje(sez)

def najhujsi_dan(okuzbe):
    if not okuzbe:
        return None
    
    d, naj_stevilo = okuzbe[0]
    naj_datum = [d]

    for datum, stevilo in okuzbe[1:]:
        if stevilo > naj_stevilo:
            naj_datum = [datum]
            naj_stevilo = stevilo
        elif stevilo == naj_stevilo:
            naj_datum.append(datum)

    if len(naj_datum) == 1:
        return naj_datum[0]
    else:
        return set(naj_datum)
